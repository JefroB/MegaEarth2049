# MegaEarth2049 Map Report Generator
# This script generates a simple report for an RPG Maker MV map file

# Function to generate a map report
function Generate-MapReport {
    param (
        [Parameter(Mandatory=$true)]
        [string]$mapFile,
        [string]$outputDir = "map_reports"
    )
    
    # Check if the map file exists
    if (-not (Test-Path $mapFile)) {
        Write-Host "Error: Map file $mapFile does not exist." -ForegroundColor Red
        return
    }
    
    # Create output directory if it doesn't exist
    if (-not (Test-Path $outputDir)) {
        New-Item -ItemType Directory -Path $outputDir | Out-Null
        Write-Host "Created output directory: $outputDir" -ForegroundColor Green
    }
    
    # Get map ID from filename
    $mapId = [regex]::Match($mapFile, 'Map(\d+)\.json').Groups[1].Value
    if (-not $mapId) {
        $mapId = "unknown"
    }
    
    Write-Host "Generating report for map $mapId..." -ForegroundColor Cyan
    
    try {
        # Read the map file
        $mapContent = Get-Content $mapFile -Raw | ConvertFrom-Json
        
        # Create a report file
        $reportFile = "$outputDir/Map${mapId}_report.txt"
        
        # Write report header
        "Map Report for Map $mapId" | Out-File -FilePath $reportFile -Encoding utf8
        "Generated on: $(Get-Date)" | Out-File -FilePath $reportFile -Encoding utf8 -Append
        "----------------------------------------" | Out-File -FilePath $reportFile -Encoding utf8 -Append
        "" | Out-File -FilePath $reportFile -Encoding utf8 -Append
        
        # Basic map information
        "Basic Information:" | Out-File -FilePath $reportFile -Encoding utf8 -Append
        "  File: $mapFile" | Out-File -FilePath $reportFile -Encoding utf8 -Append
        "  Display Name: $($mapContent.displayName)" | Out-File -FilePath $reportFile -Encoding utf8 -Append
        "  Dimensions: $($mapContent.width) x $($mapContent.height)" | Out-File -FilePath $reportFile -Encoding utf8 -Append
        "  Tileset ID: $($mapContent.tilesetId)" | Out-File -FilePath $reportFile -Encoding utf8 -Append
        "" | Out-File -FilePath $reportFile -Encoding utf8 -Append
        
        # Data array information
        "Data Array Information:" | Out-File -FilePath $reportFile -Encoding utf8 -Append
        "  Total Tiles: $($mapContent.data.Count)" | Out-File -FilePath $reportFile -Encoding utf8 -Append
        "  Expected Tiles: $($mapContent.width * $mapContent.height * 6)" | Out-File -FilePath $reportFile -Encoding utf8 -Append
        "" | Out-File -FilePath $reportFile -Encoding utf8 -Append
        
        # Events information
        "Events Information:" | Out-File -FilePath $reportFile -Encoding utf8 -Append
        
        # Count non-null events
        $eventCount = 0
        foreach ($event in $mapContent.events) {
            if ($event -ne $null) {
                $eventCount++
            }
        }
        "  Total Events: $eventCount" | Out-File -FilePath $reportFile -Encoding utf8 -Append
        
        # List events
        if ($eventCount -gt 0) {
            "  Event List:" | Out-File -FilePath $reportFile -Encoding utf8 -Append
            foreach ($event in $mapContent.events) {
                if ($event -ne $null) {
                    "    ID $($event.id): $($event.name) at ($($event.x), $($event.y))" | Out-File -FilePath $reportFile -Encoding utf8 -Append
                }
            }
        }
        
        Write-Host "  Created report at $reportFile" -ForegroundColor Green
        Write-Host "Completed generating report for map $mapId" -ForegroundColor Green
    }
    catch {
        Write-Host "Error generating report for map $mapId: $_" -ForegroundColor Red
    }
}

# Main script execution
# Example usage:
# Generate-MapReport -mapFile "data/Map001.json"

# If script is run directly (not imported as a module)
if ($MyInvocation.InvocationName -eq $MyInvocation.MyCommand.Name) {
    # Check if a map file was specified
    if ($args.Count -gt 0) {
        Generate-MapReport -mapFile $args[0]
    }
    else {
        Write-Host "Usage: .\map_report.ps1 <map_file>" -ForegroundColor Yellow
        Write-Host "Example: .\map_report.ps1 data/Map001.json" -ForegroundColor Yellow
    }
}
