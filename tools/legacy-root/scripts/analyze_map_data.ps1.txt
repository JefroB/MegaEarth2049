# MegaEarth2049 Map Data Analysis Script
# This script analyzes RPG Maker MV map files

# Function to analyze a map file
function Analyze-Map {
    param (
        [string]$mapFile,
        [string]$outputDir = "map_analysis"
    )
    
    # Check if the map file exists
    if (-not (Test-Path $mapFile)) {
        Write-Host "Error: Map file $mapFile does not exist." -ForegroundColor Red
        return
    }
    
    # Create output directory if it doesn't exist
    if (-not (Test-Path $outputDir)) {
        New-Item -ItemType Directory -Path $outputDir | Out-Null
        Write-Host "Created output directory: $outputDir" -ForegroundColor Green
    }
    
    # Get map ID from filename
    $mapId = [regex]::Match($mapFile, 'Map(\d+)\.json').Groups[1].Value
    if (-not $mapId) {
        $mapId = "unknown"
    }
    
    Write-Host "Analyzing map $mapId from $mapFile..." -ForegroundColor Cyan
    
    try {
        # Read the file content as text
        $mapContent = Get-Content $mapFile -Raw
        
        # Create a summary file
        $summaryFile = "$outputDir/Map${mapId}_summary.txt"
        "Map Summary for Map $mapId" | Out-File -FilePath $summaryFile -Encoding utf8
        
        # Extract basic properties using regex
        $displayNameMatch = [regex]::Match($mapContent, '"displayName":\s*"([^"]*)"')
        if ($displayNameMatch.Success) {
            "Display Name: $($displayNameMatch.Groups[1].Value)" | Out-File -FilePath $summaryFile -Encoding utf8 -Append
        }
        
        $widthMatch = [regex]::Match($mapContent, '"width":\s*(\d+)')
        if ($widthMatch.Success) {
            "Width: $($widthMatch.Groups[1].Value)" | Out-File -FilePath $summaryFile -Encoding utf8 -Append
        }
        
        $heightMatch = [regex]::Match($mapContent, '"height":\s*(\d+)')
        if ($heightMatch.Success) {
            "Height: $($heightMatch.Groups[1].Value)" | Out-File -FilePath $summaryFile -Encoding utf8 -Append
        }
        
        $tilesetIdMatch = [regex]::Match($mapContent, '"tilesetId":\s*(\d+)')
        if ($tilesetIdMatch.Success) {
            "Tileset ID: $($tilesetIdMatch.Groups[1].Value)" | Out-File -FilePath $summaryFile -Encoding utf8 -Append
        }
        
        Write-Host "  Created summary at $summaryFile" -ForegroundColor Green
        Write-Host "Completed analyzing map $mapId" -ForegroundColor Green
    }
    catch {
        Write-Host "Error analyzing map $mapId: $_" -ForegroundColor Red
    }
}

# Function to analyze all maps in a directory
function Analyze-AllMaps {
    param (
        [string]$mapDir = "data",
        [string]$outputDir = "map_analysis"
    )
    
    # Check if the map directory exists
    if (-not (Test-Path $mapDir)) {
        Write-Host "Error: Map directory $mapDir does not exist." -ForegroundColor Red
        return
    }
    
    # Create output directory if it doesn't exist
    if (-not (Test-Path $outputDir)) {
        New-Item -ItemType Directory -Path $outputDir | Out-Null
        Write-Host "Created output directory: $outputDir" -ForegroundColor Green
    }
    
    # Find all map files
    $mapFiles = Get-ChildItem -Path $mapDir -Filter "Map*.json"
    
    Write-Host "Found $($mapFiles.Count) map files in $mapDir" -ForegroundColor Cyan
    
    # Process each map file
    foreach ($mapFile in $mapFiles) {
        Analyze-Map -mapFile $mapFile.FullName -outputDir $outputDir
    }
    
    Write-Host "Completed analyzing all maps" -ForegroundColor Green
}

# Main script execution
# Example usage:
# Analyze a single map:
# Analyze-Map -mapFile "data/Map001.json"
#
# Analyze all maps in the data directory:
# Analyze-AllMaps
